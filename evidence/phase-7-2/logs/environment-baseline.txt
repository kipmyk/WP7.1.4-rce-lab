# ==============================================================================
# Apache 2.4+ Hardening Directive: WordPress Upgrade Directory
# Target: /wp-content/upgrade/
# ==============================================================================
#
# Context: VirtualHost or Main Server Configuration (Recommended)
# Alternatively: Placed in web root .htaccess via mod_rewrite / location match
#
# Operational Rationale:
# WordPress core, theme, and plugin update operations utilize local server-side
# filesystem APIs (WP_Filesystem) to extract, stage, and copy upgrade files.
# At no point in a legitimate update does a client browser or external visitor
# need to request files from /wp-content/upgrade/ over HTTP.
#
# ==============================================================================

# Option A: Server / VirtualHost Directory Block (Recommended)
<Directory "/var/www/html/wp-content/upgrade">
    # 1. Deny all direct HTTP requests (returns 403 Forbidden)
    <IfModule mod_authz_core.c>
        Require all denied
    </IfModule>
    <IfModule !mod_authz_core.c>
        Order deny,allow
        Deny from all
    </IfModule>

    # 2. Disable directory listing explicitly
    Options -Indexes

    # 3. Prevent PHP execution if access controls are misconfigured or bypassed
    <FilesMatch "(?i)\.(php|php[0-9]?|phtml|phar|inc)$">
        <IfModule mod_authz_core.c>
            Require all denied
        </IfModule>
        <IfModule mod_php.c>
            php_flag engine off
        </IfModule>
        SetHandler none
    </FilesMatch>
</Directory>
